🚀 如何使用

Python 版本

# 克隆仓库
git clone https://github.com/webcstx/doubao-quick-tools.git
cd doubao-quick-tools

# 运行密码生成器
python3 python/password_gen.py

# 运行文本反转
python3 python/text_reverse.py

# 运行随机名言
python3 python/random_quote.py

PHP 版本

# 克隆仓库
git clone https://github.com/webcstx/doubao-quick-tools.git
cd doubao-quick-tools

# 运行密码生成器
php php/password_gen.php

# 运行文本反转
php php/text_reverse.php

# 运行随机名言
php php/random_quote.php
